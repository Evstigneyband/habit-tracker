    boot();

    function boot() {
      bindEvents();
      renderDraftGoals();
      if (state.sessionToken) {
        setLoading(true);
        server('apiGetSession', state.sessionToken)
          .then(appState => handleAppState(appState))
          .catch(() => {
            forgetToken();
            state.sessionToken = '';
            showScreen('auth');
          })
          .finally(() => setLoading(false));
      } else {
        showScreen('auth');
      }
    }

      els.loginTab.addEventListener('click', () => setAuthMode('login'));

      els.menuChallenges.addEventListener('click', () => {

        showChallenges();

        closeMenu();

    function setAuthMode(mode) {
      state.mode = mode;
      els.loginTab.classList.toggle('active', mode === 'login');
      els.registerTab.classList.toggle('active', mode === 'register');
      els.authSubmit.textContent = mode === 'login' ? 'Войти' : 'Зарегистрироваться';
      els.authPassword.autocomplete = mode === 'login' ? 'current-password' : 'new-password';
      setMessage(els.authMessage, '');
    }

    function handleAuth(event) {
      event.preventDefault();
      setMessage(els.authMessage, '');
      setLoading(true);
      const method = state.mode === 'login' ? 'apiLogin' : 'apiRegister';
      server(method, {
        email: els.authEmail.value,
        password: els.authPassword.value,
      })
        .then(appState => handleAppState(appState))
        .catch(error => setMessage(els.authMessage, error.message, true))
        .finally(() => setLoading(false));
    }

    function handleLogout() {
      const token = state.sessionToken;
      stopAutoRefresh();
      closeMenu();
      forgetToken();
      state.sessionToken = '';
      state.app = null;
      state.currentChallengeId = '';
      state.editingChallengeId = '';
      state.editReturnChallengeId = '';
      state.openChallengeActionsId = '';
      state.pendingSaves = 0;
      state.pendingGoalUpdates = {};
      state.goalRequestSeq = {};
      setAuthMode('login');
      els.menuWrap.hidden = true;
      els.bottomNav.hidden = true;
      els.authForm.reset();
      els.authEmail.value = '';
      els.authPassword.value = '';
      els.challengesList.innerHTML = '';
      els.simpleDashboardList.innerHTML = '';
      els.timeDashboardList.innerHTML = '';
      els.analyticsMetrics.innerHTML = '';
      els.analyticsDays.innerHTML = '';
      els.analyticsGoals.innerHTML = '';
      els.analyticsInsights.innerHTML = '';
      setMessage(els.authMessage, '');
      showScreen('auth');
      setTimeout(() => {
        els.authEmail.value = '';
        els.authPassword.value = '';
      }, 60);
      if (token) server('apiLogout', token).catch(() => {});
    }

    function addSimpleGoal() {
      const title = els.simpleGoalInput.value.trim();
      if (!title) return;
      state.draftSimpleGoals.push({ title });
      els.simpleGoalInput.value = '';
      renderDraftGoals();
    }

    function addTimeGoal() {
      const title = els.timeGoalInput.value.trim();
      const targetHours = Number(els.timeGoalHours.value || 0);
      if (!title || targetHours <= 0) return;
      state.draftTimeGoals.push({ title, targetHours });
      els.timeGoalInput.value = '';
      els.timeGoalHours.value = '2';
      renderDraftGoals();
    }

    function removeDraftGoal(type, index) {
      if (type === 'simple') state.draftSimpleGoals.splice(index, 1);
      if (type === 'time') state.draftTimeGoals.splice(index, 1);
      renderDraftGoals();
    }

    function updateDraftGoal(type, index, field, value) {
      const collection = type === 'simple' ? state.draftSimpleGoals : state.draftTimeGoals;
      if (!collection[index]) return;
      collection[index][field] = field === 'targetHours' ? Number(value || 0) : value;
      renderDraftGoals();
    }

    function moveDraftGoal(type, index, direction) {
      const collection = type === 'simple' ? state.draftSimpleGoals : state.draftTimeGoals;
      const nextIndex = index + direction;
      if (nextIndex < 0 || nextIndex >= collection.length) return;
      const item = collection[index];
      collection.splice(index, 1);
      collection.splice(nextIndex, 0, item);
      renderDraftGoals();
    }

    function renderDraftGoals() {
      els.simpleGoalsList.innerHTML = state.draftSimpleGoals.length
        ? state.draftSimpleGoals.map((goal, index) => draftGoalEditor('simple', goal, index)).join('')
        : '<div class="empty">Добавь первую цель без часов</div>';
      els.timeGoalsList.innerHTML = state.draftTimeGoals.length
        ? state.draftTimeGoals.map((goal, index) => draftGoalEditor('time', goal, index)).join('')
        : '<div class="empty">Добавь первую цель с часами</div>';
    }

    function draftGoalEditor(type, goal, index) {
      const isTime = type === 'time';
      return `
        <div class="pill-item">
          <div class="${isTime ? 'draft-time-row' : 'draft-edit'}">
            <input class="input" type="text" value="${escapeAttribute(goal.title)}" onchange="updateDraftGoal('${type}', ${index}, 'title', this.value)">
            ${isTime ? `<input class="input" type="number" min="0.5" max="24" step="0.5" value="${Number(goal.targetHours || 0)}" onchange="updateDraftGoal('${type}', ${index}, 'targetHours', this.value)">` : ''}
          </div>
          <div class="draft-actions">
            <button class="mini-button" type="button" onclick="moveDraftGoal('${type}', ${index}, -1)" aria-label="Выше">↑</button>
            <button class="mini-button" type="button" onclick="moveDraftGoal('${type}', ${index}, 1)" aria-label="Ниже">↓</button>
            <button class="remove-button" type="button" onclick="removeDraftGoal('${type}', ${index})" aria-label="Удалить">x</button>
          </div>
        </div>
      `;
    }

    function handleCreateChallenge(event) {
      event.preventDefault();
      setMessage(els.setupMessage, '');
      if (!state.draftSimpleGoals.length && !state.draftTimeGoals.length) {
        setMessage(els.setupMessage, 'Добавь хотя бы одну цель.', true);
        return;
      }
      setLoading(true);
      const payload = {
        title: els.challengeTitle.value,
        durationDays: Number(els.challengeDays.value),
        simpleGoals: state.draftSimpleGoals,
        timeGoals: state.draftTimeGoals,
      };
      const request = state.editingChallengeId
        ? server('apiRestartChallenge', state.sessionToken, state.editingChallengeId, payload)
        : server('apiCreateChallenge', state.sessionToken, payload);
      request
        .then(appState => {
          state.editReturnChallengeId = '';
          resetChallengeForm();
          handleAppState(appState);
        })
        .catch(error => setMessage(els.setupMessage, error.message, true))
        .finally(() => setLoading(false));
    }

    function handleAppState(appState) {
      if (!appState || typeof appState !== 'object') {
        throw new Error('Apps Script вернул пустой ответ. Обнови развертывание и попробуй еще раз.');
      }
      appState = mergePendingGoalUpdates(appState);
      state.app = appState;
      state.sessionToken = appState.sessionToken;
      rememberToken(appState.sessionToken);
      els.menuWrap.hidden = false;
      els.userCaption.textContent = appState.user.email;
      state.restoringChallengeId = '';
      state.currentChallengeId = appState.challenge ? appState.challenge.challengeId : '';
      rememberChallengeId(appState.user, state.currentChallengeId);
      updateMenuState();
      renderChallenges(appState);
      els.setupActions.hidden = !(appState.challenges && appState.challenges.length);
      if (!appState.challenge) {
        showScreen('setup');
        return;
      }
      renderDashboard(appState);
      if (appState.challenge.isFinished) {
        showAnalytics();
      } else {
        showScreen('dashboard');
      }
      startAutoRefresh();
    }

    function bindChallengeRows() {
      const rows = Array.from(els.challengesList.querySelectorAll('.challenge-row'));
      rows.forEach(row => {
        const challengeId = row.dataset.challengeId;
        const item = row.querySelector('.challenge-item');
        const editButton = row.querySelector('[data-action="edit"]');
        const deleteButton = row.querySelector('[data-action="delete"]');
        let startX = 0;
        let currentX = 0;
        let isDragging = false;
        let didSwipe = false;

        item.addEventListener('pointerdown', event => {
          startX = event.clientX;
          currentX = startX;
          isDragging = true;
          didSwipe = false;
          item.setPointerCapture(event.pointerId);
          item.style.transition = 'none';
        });

        item.addEventListener('pointermove', event => {
          if (!isDragging) return;
          currentX = event.clientX;
          const deltaX = currentX - startX;
          if (Math.abs(deltaX) < 8) return;
          if (deltaX < 0) {
            didSwipe = true;
            item.style.transform = `translateX(${Math.max(deltaX, -104)}px)`;
          } else if (row.classList.contains('open')) {
            didSwipe = true;
            item.style.transform = `translateX(${Math.min(-104 + deltaX, 0)}px)`;
          }
        });

        item.addEventListener('pointerup', () => {
          if (!isDragging) return;
          isDragging = false;
          item.style.transition = '';
          item.style.transform = '';
          const deltaX = currentX - startX;
          if (deltaX < -44) {
            closeChallengeRows(row);
            row.classList.add('open');
            state.openChallengeActionsId = challengeId;
            return;
          }
          if (deltaX > 36) {
            row.classList.remove('open');
            if (state.openChallengeActionsId === challengeId) state.openChallengeActionsId = '';
            return;
          }
          if (didSwipe) {
            const shouldOpen = deltaX < 0;
            if (shouldOpen) closeChallengeRows(row);
            row.classList.toggle('open', shouldOpen);
            state.openChallengeActionsId = shouldOpen ? challengeId : '';
          }
        });

        item.addEventListener('pointercancel', () => {
          isDragging = false;
          item.style.transition = '';
          item.style.transform = '';
        });

        item.addEventListener('click', event => {
          if (didSwipe) {
            event.preventDefault();
            didSwipe = false;
            return;
          }
          if (row.classList.contains('open')) {
            row.classList.remove('open');
            if (state.openChallengeActionsId === challengeId) state.openChallengeActionsId = '';
            return;
          }
          state.openChallengeActionsId = '';
          openChallenge(challengeId);
        });

        bindChallengeAction(editButton, () => editChallenge(challengeId));
        bindChallengeAction(deleteButton, () => deleteChallenge(challengeId));
      });
    }

    function bindChallengeAction(button, action) {
      let handledAt = 0;
      const run = event => {
        event.preventDefault();
        event.stopPropagation();
        const now = Date.now();
        if (now - handledAt < 450) return;
        handledAt = now;
        state.openChallengeActionsId = '';
        closeChallengeRows();
        action();
      };
      button.addEventListener('pointerup', run);
      button.addEventListener('click', event => {
        if (Date.now() - handledAt < 450) {
          event.preventDefault();
          event.stopPropagation();
          return;
        }
        run(event);
      });
    }

    function closeChallengeRows(exceptRow) {
      els.challengesList.querySelectorAll('.challenge-row.open').forEach(row => {
        if (row !== exceptRow) row.classList.remove('open');
      });
      if (!exceptRow) state.openChallengeActionsId = '';
    }

    function renderDashboard(appState) {
      const challenge = appState.challenge;
      const progress = appState.progress;
      const overallProgress = appState.overallProgress || { percent: 0, elapsedDays: 0, durationDays: challenge.durationDays };
      els.challengeName.textContent = challenge.title;
      els.challengeMeta.textContent = `День ${challenge.dayNumber} из ${challenge.durationDays}`;
      els.progressPercent.textContent = `${progress.percent}%`;
      els.progressFill.style.width = `${progress.percent}%`;
      els.todayProgressLabel.textContent = `${progress.percent}%`;
      els.overallProgressLabel.textContent = `${overallProgress.percent}%`;
      els.overallProgressFill.style.width = `${overallProgress.percent}%`;
      els.progressCaption.textContent = `${progress.completed} из ${progress.total} целей закрыто сегодня. Общий прогресс: ${overallProgress.percent}%`;

      const simpleGoals = appState.goals.filter(goal => goal.goalType === 'simple');
      const timeGoals = appState.goals.filter(goal => goal.goalType === 'time');
      els.simpleDashboardList.innerHTML = simpleGoals.length
        ? simpleGoals.map(renderSimpleGoal).join('')
        : '<div class="empty">Целей без часов нет</div>';
      els.timeDashboardList.innerHTML = timeGoals.length
        ? timeGoals.map(renderTimeGoal).join('')
        : '<div class="empty">Целей с часами нет</div>';
    }

    function renderAnalytics(appState) {
      const challenge = appState.challenge;
      const analytics = appState.analytics;
      if (!challenge || !analytics) return;
      const summary = analytics.summary || {};
      els.analyticsTitle.textContent = challenge.title;
      els.analyticsLead.textContent = challenge.isFinished
        ? `Челлендж завершён. ${summary.elapsedDays || 0} из ${summary.durationDays || challenge.durationDays} дней в истории.`
        : `День ${challenge.dayNumber} из ${challenge.durationDays}. Аналитика обновляется по уже пройденным дням.`;
      els.analyticsToday.hidden = Boolean(challenge.isFinished);
      els.analyticsMetrics.innerHTML = [
        metric('Общий прогресс', `${summary.overallPercent || 0}%`),
        metric('Средний день', `${summary.averageDayPercent || 0}%`),
        metric('Дней на 100%', String(summary.completedDays || 0)),
        metric('Дней ниже 50%', String(summary.lowDays || 0)),
      ].join('');
      els.analyticsDays.innerHTML = (analytics.days || []).map(renderAnalyticsDay).join('');
      els.analyticsGoals.innerHTML = (analytics.goals || []).length
        ? analytics.goals.map(renderAnalyticsGoal).join('')
        : '<div class="empty">Целей пока нет</div>';
      els.analyticsInsights.innerHTML = renderInsights(summary);
    }

    function metric(label, value) {
      return `<div class="metric"><strong>${escapeHtml(value)}</strong><span>${escapeHtml(label)}</span></div>`;
    }

    function renderAnalyticsDay(day) {
      const tone = day.isFuture ? 'future' : day.percent === 100 ? 'good' : day.percent >= 50 ? 'mid' : 'low';
      return `
        <div class="day-cell ${tone}" title="${escapeAttribute(formatDate(day.date))}: ${day.percent}%">
          <span>${day.dayNumber}</span>
          <small>${day.isFuture ? '' : `${day.percent}%`}</small>
        </div>
      `;
    }

    function renderAnalyticsGoal(goal) {

        ? `Факт ${formatHours(goal.actualHours)} из ${formatHours(goal.plannedHours)}`
        : `${goal.completedCount} из ${goal.elapsedDays} дней`;
      return `
        <article class="analytics-goal">
          <div class="analytics-goal-head">
            <span>${escapeHtml(goal.title)}</span>
            <span>${goal.completionPercent}%</span>
          </div>
          <div class="mini-track"><div class="mini-fill" style="width:${goal.completionPercent}%"></div></div>
          <div class="state-line">${escapeHtml(meta)}</div>
        </article>
      `;
    }

    function renderInsights(summary) {
      const insights = [];
      if (summary.strongestGoal) insights.push(`Стабильнее всего: ${summary.strongestGoal.title} (${summary.strongestGoal.completionPercent}%).`);
      if (summary.weakestGoal) insights.push(`Больше всего проседает: ${summary.weakestGoal.title} (${summary.weakestGoal.completionPercent}%).`);
      if (summary.bestDay) insights.push(`Лучший день: ${formatDate(summary.bestDay.date)}, ${summary.bestDay.percent}%.`);
      if (summary.worstDay) insights.push(`Самый слабый день: ${formatDate(summary.worstDay.date)}, ${summary.worstDay.percent}%.`);
      if (!insights.length) insights.push('Данных пока мало. Отметь хотя бы один день, и здесь появятся выводы.');
      return insights.map(text => `<div class="insight">${escapeHtml(text)}</div>`).join('');
    }

    function toggleSimple(goalId, isChecked) {
      setMessage(els.dashboardMessage, '');
      const patch = {
        isChecked,
        isCompleted: isChecked,
      };
      const requestSeq = queueGoalUpdate(goalId, patch);
      state.pendingSaves += 1;
      server('apiUpdateSimpleEntry', state.sessionToken, { goalId, isChecked })
        .then(appState => handleGoalSaveSuccess(goalId, requestSeq, appState))
        .catch(error => handleGoalSaveError(goalId, requestSeq, error))
        .finally(() => {
          state.pendingSaves = Math.max(0, state.pendingSaves - 1);
        });
    }

    function setHours(goalId, actualHours) {
      const goal = state.app.goals.find(item => item.goalId === goalId);
      if (!goal) return;
      setMessage(els.dashboardMessage, '');
      const patch = {
        actualHours,
        isCompleted: Number(actualHours || 0) >= Number(goal.targetHours || 0),
      };
      const requestSeq = queueGoalUpdate(goalId, patch);
      state.pendingSaves += 1;
      server('apiUpdateTimeEntry', state.sessionToken, { goalId, actualHours })
        .then(appState => handleGoalSaveSuccess(goalId, requestSeq, appState))
        .catch(error => handleGoalSaveError(goalId, requestSeq, error))
        .finally(() => {
          state.pendingSaves = Math.max(0, state.pendingSaves - 1);
        });
    }

    function queueGoalUpdate(goalId, patch) {
      const nextSeq = (state.goalRequestSeq[goalId] || 0) + 1;
      state.goalRequestSeq[goalId] = nextSeq;
      state.pendingGoalUpdates[goalId] = { patch, seq: nextSeq };
      applyLocalGoalUpdate(goalId, patch);
      return nextSeq;
    }

    function handleGoalSaveSuccess(goalId, requestSeq, appState) {
      if (!isLatestGoalRequest(goalId, requestSeq)) return;
      delete state.pendingGoalUpdates[goalId];
      handleAppState(appState);
    }

    function handleGoalSaveError(goalId, requestSeq, error) {
      if (!isLatestGoalRequest(goalId, requestSeq)) return;
      delete state.pendingGoalUpdates[goalId];
      setMessage(els.dashboardMessage, error.message || 'Не удалось сохранить. Сейчас обновлю данные.', true);
      refreshCurrentChallenge(true);
    }

    function isLatestGoalRequest(goalId, requestSeq) {
      return state.goalRequestSeq[goalId] === requestSeq;
    }

    function applyLocalGoalUpdate(goalId, patch) {
      if (!state.app || !state.app.goals) return;
      state.app.goals = state.app.goals.map(goal => {
        if (goal.goalId !== goalId) return goal;
        return Object.assign({}, goal, patch);
      });
      recalculateProgress();
      renderDashboard(state.app);
    }

    function recalculateProgress() {
      state.app.progress = calculateProgress(state.app.goals || []);
    }

    function calculateProgress(goals) {
      const total = goals.length;
      const completed = goals.filter(goal => goal.isCompleted).length;
      return {
        completed,
        total,
        percent: total ? Math.round((completed / total) * 100) : 0,
      };
    }

    function mergePendingGoalUpdates(appState) {
      const pendingGoalIds = Object.keys(state.pendingGoalUpdates);
      if (!pendingGoalIds.length || !appState.goals) return appState;
      const pending = state.pendingGoalUpdates;
      appState.goals = appState.goals.map(goal => {
        const pendingUpdate = pending[goal.goalId];
        if (!pendingUpdate) return goal;
        return Object.assign({}, goal, pendingUpdate.patch);
      });
      appState.progress = calculateProgress(appState.goals);
      if (appState.challenges && appState.challenge) {
        appState.challenges = appState.challenges.map(challenge => (
          challenge.challengeId === appState.challenge.challengeId
            ? Object.assign({}, challenge, { isCurrent: true })
            : Object.assign({}, challenge, { isCurrent: false })
        ));
      }
      return appState;
    }

    function showChallenges() {
      if (!state.sessionToken || !state.app) {
        showScreen('auth');
        return;
      }
      if (restoreEditReturn('challenges')) return;
      if (state.app) renderChallenges(state.app);
      showScreen('challenges');
      refreshCurrentChallenge();
    }

    function showAnalytics() {
      if (!state.sessionToken || !state.app) {
        showScreen('auth');
        return;
      }
      if (restoreEditReturn('analytics')) return;
      if (!state.app || !state.app.challenge) return;
      renderAnalytics(state.app);
      showScreen('analytics');
    }

    function startNewChallenge() {
      if (!state.sessionToken || !state.app) {
        showScreen('auth');
        return;
      }
      state.editReturnChallengeId = '';
      resetChallengeForm();
      setMessage(els.setupMessage, '');
      showScreen('setup');
      setTimeout(() => els.challengeTitle.focus(), 60);
    }

    function startEditChallenge() {
      if (!state.app || !state.app.challenge) return;
      stopAutoRefresh();
      if (!state.editReturnChallengeId) state.editReturnChallengeId = state.currentChallengeId || state.app.challenge.challengeId;
      state.editingChallengeId = state.app.challenge.challengeId;
      els.setupTitle.textContent = 'Редактируй челлендж.';
      els.setupLead.textContent = 'После сохранения челлендж начнется заново с сегодняшней даты, а текущие отметки сбросятся.';
      els.challengeSubmit.textContent = 'Перезапустить челлендж';
      els.challengeTitle.value = state.app.challenge.title || '';
      els.challengeDays.value = state.app.challenge.durationDays || 30;
      state.draftSimpleGoals = (state.app.goals || [])
        .filter(goal => goal.goalType === 'simple')
        .map(goal => ({ title: goal.title }));
      state.draftTimeGoals = (state.app.goals || [])
        .filter(goal => goal.goalType === 'time')
        .map(goal => ({ title: goal.title, targetHours: Number(goal.targetHours || 0) }));
      renderDraftGoals();
      setMessage(els.setupMessage, '');
      showScreen('setup');
      setTimeout(() => els.challengeTitle.focus(), 60);
    }

    function editChallenge(challengeId) {
      stopAutoRefresh();
      state.editReturnChallengeId = state.currentChallengeId || (state.app && state.app.challenge && state.app.challenge.challengeId) || '';
      state.currentChallengeId = challengeId;
      const openSeq = ++state.challengeOpenSeq;
      setLoading(true);
      server('apiPreviewChallenge', state.sessionToken, challengeId)
        .then(appState => {
          if (openSeq !== state.challengeOpenSeq) return;
          if (!appState || !appState.challenge || appState.challenge.challengeId !== state.currentChallengeId) return;
          appState = mergePendingGoalUpdates(appState);
          state.app = appState;
          state.sessionToken = appState.sessionToken;
          state.currentChallengeId = appState.challenge.challengeId;
          rememberToken(appState.sessionToken);
          els.userCaption.textContent = appState.user.email;
          renderChallenges(appState);
          els.setupActions.hidden = !(appState.challenges && appState.challenges.length);
          startEditChallenge();
        })
        .catch(error => setMessage(els.setupMessage, error.message, true))
        .finally(() => setLoading(false));
    }

    function resetChallengeForm() {
      state.editingChallengeId = '';
      state.editReturnChallengeId = '';
      state.draftSimpleGoals = [];
      state.draftTimeGoals = [];
      els.challengeForm.reset();
      els.challengeDays.value = '30';
      els.setupTitle.textContent = 'Создай челлендж.';
      els.setupLead.textContent = 'Все цели получают одинаковый вес. Если целей десять, каждая закрытая цель добавляет 10% к дню.';
      els.challengeSubmit.textContent = 'Начать челлендж';
      renderDraftGoals();
      setMessage(els.setupMessage, '');
    }

    function deleteChallenge(challengeId) {
      const challenge = state.app && state.app.challenges
        ? state.app.challenges.find(item => item.challengeId === challengeId)
        : null;
      const title = challenge ? challenge.title : 'этот челлендж';
      if (!confirm(`Удалить челлендж «${title}»? Это действие нельзя отменить.`)) return;
      stopAutoRefresh();
      setLoading(true);
      server('apiDeleteChallenge', state.sessionToken, challengeId, state.currentChallengeId)
        .then(appState => {
          state.currentChallengeId = appState.challenge ? appState.challenge.challengeId : '';
          handleAppState(appState);
        })
        .catch(error => setMessage(els.setupMessage, error.message, true))
        .finally(() => setLoading(false));
    }

    function openLatestChallenge() {
      const latest = state.app && state.app.challenges && state.app.challenges[0];
      if (latest) openChallenge(latest.challengeId);
    }

    function openCurrentChallenge() {
      if (!state.sessionToken || !state.app) {
        showScreen('auth');
        return;
      }
      if (restoreEditReturn('dashboard')) return;
      if (state.app && state.app.challenge) {
        showScreen('dashboard');
        return;
      }
      openLatestChallenge();
    }

    function restoreEditReturn(targetScreen) {
      if (!state.editingChallengeId) return false;
      const returnChallengeId = state.editReturnChallengeId || state.currentChallengeId;
      const editingChallengeId = state.editingChallengeId;
      if (!returnChallengeId || returnChallengeId === editingChallengeId) {
        resetChallengeForm();
        if (targetScreen === 'analytics') showAnalytics();
        else if (targetScreen === 'challenges') showChallenges();
        else showScreen('dashboard');
        return true;
      }

      resetChallengeForm();
      stopAutoRefresh();
      state.currentChallengeId = returnChallengeId;
      const openSeq = ++state.challengeOpenSeq;
      setLoading(true);
      server('apiOpenChallenge', state.sessionToken, returnChallengeId)
        .then(appState => {
          if (openSeq !== state.challengeOpenSeq) return;
          if (!appState || !appState.challenge || appState.challenge.challengeId !== state.currentChallengeId) return;
          handleAppState(appState);
          if (targetScreen === 'analytics') showAnalytics();
          if (targetScreen === 'challenges') showChallenges();
        })
        .catch(error => setMessage(els.setupMessage, error.message, true))
        .finally(() => setLoading(false));
      return true;
    }

    function openChallenge(challengeId) {
      stopAutoRefresh();
      state.currentChallengeId = challengeId;
      const openSeq = ++state.challengeOpenSeq;
      setLoading(true);
      server('apiOpenChallenge', state.sessionToken, challengeId)
        .then(appState => {
          if (openSeq !== state.challengeOpenSeq) return;
          if (!appState || !appState.challenge || appState.challenge.challengeId !== state.currentChallengeId) return;
          handleAppState(appState);
        })
        .catch(error => setMessage(els.setupMessage, error.message, true))
        .finally(() => setLoading(false));
    }

    function startAutoRefresh() {
      stopAutoRefresh();
      state.refreshTimer = setInterval(refreshCurrentChallenge, REFRESH_INTERVAL_MS);
    }

    function stopAutoRefresh() {
      if (!state.refreshTimer) return;
      clearInterval(state.refreshTimer);
      state.refreshTimer = null;
    }

    function refreshCurrentChallenge(force) {
      const challengeId = state.currentChallengeId || (state.app && state.app.challenge && state.app.challenge.challengeId);
      if (!state.sessionToken || !challengeId) return;
      if (!force && (document.hidden || state.pendingSaves > 0 || state.refreshInFlight)) return;
      state.refreshInFlight = true;
      server('apiOpenChallenge', state.sessionToken, challengeId)
        .then(appState => applyRemoteState(appState))
        .catch(() => {})
        .finally(() => {
          state.refreshInFlight = false;
        });
    }

    function applyRemoteState(appState) {
      if (!appState || typeof appState !== 'object') return;
      if (state.currentChallengeId && (!appState.challenge || appState.challenge.challengeId !== state.currentChallengeId)) return;
      appState = mergePendingGoalUpdates(appState);
      state.app = appState;
      state.sessionToken = appState.sessionToken;
      state.currentChallengeId = appState.challenge ? appState.challenge.challengeId : state.currentChallengeId;
      rememberToken(appState.sessionToken);
      renderChallenges(appState);
      els.setupActions.hidden = !(appState.challenges && appState.challenges.length);
      if (state.activeScreen === 'dashboard' && appState.challenge) renderDashboard(appState);
      if (state.activeScreen === 'analytics' && appState.challenge) renderAnalytics(appState);
      if (state.activeScreen === 'challenges') renderChallenges(appState);
    }

    function showScreen(name) {
      state.activeScreen = name;
      els.authScreen.classList.toggle('active', name === 'auth');
      els.setupScreen.classList.toggle('active', name === 'setup');
      els.challengesScreen.classList.toggle('active', name === 'challenges');
      els.dashboardScreen.classList.toggle('active', name === 'dashboard');
      els.analyticsScreen.classList.toggle('active', name === 'analytics');
      els.menuWrap.hidden = name === 'auth';
      els.bottomNav.hidden = name === 'auth';
      if (name === 'auth') els.userCaption.textContent = 'Личный трекер прогресса';
      if (name === 'auth') {
        closeMenu();
        stopAutoRefresh();
      }
      updateMenuState();
      scrollToScreenTop();
    }

    function scrollToScreenTop() {
      requestAnimationFrame(() => {
        window.scrollTo({ top: 0, left: 0, behavior: 'auto' });
        document.documentElement.scrollTop = 0;
        document.body.scrollTop = 0;
      });
    }

    function updateMenuState() {
      const insideChallenge = (state.activeScreen === 'dashboard' || state.activeScreen === 'analytics') && state.app && state.app.challenge;
      els.menuAnalytics.hidden = !insideChallenge;
      els.menuEdit.hidden = !insideChallenge;
      els.navToday.classList.toggle('active', state.activeScreen === 'dashboard');
      els.navChallenges.classList.toggle('active', state.activeScreen === 'challenges');
      els.navAnalytics.classList.toggle('active', state.activeScreen === 'analytics');
      els.navProfile.classList.remove('active');
      els.navAnalytics.disabled = !(state.app && state.app.challenge);
    }

    function toggleMenu(event) {
      event.stopPropagation();
      const isOpen = els.menuPanel.classList.toggle('open');
      els.menuButton.setAttribute('aria-expanded', String(isOpen));
    }

    function closeMenu() {
      els.menuPanel.classList.remove('open');
      els.menuButton.setAttribute('aria-expanded', 'false');
    }

    function setLoading(isLoading) {
      els.loading.classList.toggle('active', isLoading);
    }

    function setMessage(element, message, isError) {
      element.textContent = message || '';
      element.classList.toggle('error', Boolean(isError));
    }

    function getStoredToken() {
      return getLocalToken() || getCookieToken();
    }

    function getLocalToken() {
      try {
        return localStorage.getItem(TOKEN_KEY) || '';
      } catch (error) {
        return '';
      }
    }

    function getCookieToken() {
      const match = document.cookie.match(new RegExp('(?:^|; )' + TOKEN_KEY + '=([^;]*)'));
      return match ? decodeURIComponent(match[1]) : '';
    }

    function rememberToken(token) {
      if (!token) return;
      try {
        localStorage.setItem(TOKEN_KEY, token);
      } catch (error) {}
      document.cookie = `${TOKEN_KEY}=${encodeURIComponent(token)}; max-age=${TOKEN_MAX_AGE}; path=/; SameSite=Lax; Secure`;
    }

    function forgetToken() {
      try {
        localStorage.removeItem(TOKEN_KEY);
      } catch (error) {}
      document.cookie = `${TOKEN_KEY}=; max-age=0; path=/; SameSite=Lax; Secure`;
    }

    function challengeStorageKey(user) {
      const userKey = user && (user.userId || user.email);
      return LAST_CHALLENGE_PREFIX + String(userKey || 'default');
    }

    function getStoredChallengeId(user) {
      try {
        return localStorage.getItem(challengeStorageKey(user)) || '';
      } catch (error) {
        return '';
      }
    }

    function rememberChallengeId(user, challengeId) {
      if (!challengeId) return;
      try {
        localStorage.setItem(challengeStorageKey(user), challengeId);
      } catch (error) {}
    }

    function forgetChallengeId(user) {
      try {
        localStorage.removeItem(challengeStorageKey(user));
      } catch (error) {}
    }

    function server(method, ...args) {
      return new Promise((resolve, reject) => {
        google.script.run
          .withSuccessHandler(resolve)
          .withFailureHandler(error => {
            reject(new Error(error && error.message ? error.message : String(error || 'Неизвестная ошибка Apps Script.')));
          })[method](...args);
      });
    }

    function formatHours(value) {
      const number = Number(value || 0);
      if (number === 0) return '0';
      if (number === 0.5) return '30 минут';
      if (number % 1 === 0.5) {
        const whole = Math.floor(number);
        return `${whole} ${hourWord(whole)} 30 минут`;
      }
      if (number === 1) return '1 час';
      if (number > 1 && number < 5) return `${number} часа`;
      return `${number} часов`;
    }

    function hourWord(value) {
      if (value === 1) return 'час';
      if (value > 1 && value < 5) return 'часа';
      return 'часов';
    }

    function buildTimeOptions() {
      const options = [];
      for (let value = 0; value <= 10; value += 0.5) {
        options.push(value);
      }
      return options;
    }

    function formatDate(value) {
      const parts = String(value || '').slice(0, 10).split('-');
      if (parts.length !== 3) return value || '';
      return `${parts[2]}.${parts[1]}.${parts[0]}`;
    }

    function escapeHtml(value) {
      return String(value)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }

    function escapeAttribute(value) {
      return escapeHtml(value).replace(/`/g, '&#096;');
    }
